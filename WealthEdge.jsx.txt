function uid() { return Math.random().toString(36).slice(2,10); }

function calcRR(entry, sl, tp, side) {
  const e = parseFloat(entry), s = parseFloat(sl), t = parseFloat(tp);
  if (!e || !s || !t) return null;
  const risk   = side === "Long" ? e - s : s - e;
  const reward = side === "Long" ? t - e : e - t;
  if (risk <= 0 || reward <= 0) return null;
  return (reward / risk).toFixed(2);
}

function calcPnl(entry, exit, qty, side) {
  const e = parseFloat(entry), x = parseFloat(exit), q = parseFloat(qty);
  if (!e || !x || !q) return null;
  return side === "Long" ? ((x - e) * q).toFixed(2) : ((e - x) * q).toFixed(2);
}

// ─── TINY SPARKLINE ─────────────────────────────────────────────────────────

function Sparkline({ data, color, width = 120, height = 36 }) {
  if (!data || data.length < 2) return null;
  const min = Math.min(...data), max = Math.max(...data);
  const range = max - min || 1;
  const pts = data.map((v, i) => {
    const x = (i / (data.length - 1)) * width;
    const y = height - ((v - min) / range) * height;
    return `${x},${y}`;
  }).join(" ");
  return (
    <svg width={width} height={height} style={{ overflow:"visible" }}>
      <polyline points={pts} fill="none" stroke={color} strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
      <polyline points={`0,${height} ${pts} ${width},${height}`} fill={`url(#sg${color.replace("#","")})`} stroke="none" />
      <defs>
        <linearGradient id={`sg${color.replace("#","")}`} x1="0" y1="0" x2="0" y2="1">
          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
          <stop offset="100%" stopColor={color} stopOpacity="0" />
        </linearGradient>
      </defs>
    </svg>
  );
}

// ─── DONUT CHART ────────────────────────────────────────────────────────────

function Donut({ pct, color, size = 72, label }) {
  const r = (size - 10) / 2, cx = size / 2, cy = size / 2;
  const circ = 2 * Math.PI * r;
  const dash = (pct / 100) * circ;
  return (
    <div style={{ display:"flex", flexDirection:"column", alignItems:"center", gap:4 }}>
      <svg width={size} height={size}>
        <circle cx={cx} cy={cy} r={r} fill="none" stroke="rgba(255,255,255,.08)" strokeWidth="7" />
        <circle cx={cx} cy={cy} r={r} fill="none" stroke={color} strokeWidth="7"
          strokeDasharray={`${dash} ${circ}`} strokeLinecap="round"
          transform={`rotate(-90 ${cx} ${cy})`} style={{ transition:"stroke-dasharray .6s ease" }} />
        <text x={cx} y={cy + 5} textAnchor="middle" fill="white" fontSize="13" fontWeight="700">{pct}%</text>
      </svg>
      {label && <span style={{ fontSize:11, opacity:.6 }}>{label}</span>}
    </div>
  );
}

function CalendarView({ trades, accent }) {
  const now = new Date();
  const [year, setYear] = useState(now.getFullYear());
  const [month, setMonth] = useState(now.getMonth());

  const first = new Date(year, month, 1).getDay();
  const days = new Date(year, month + 1, 0).getDate();

  const byDay = {};
  trades.forEach(t => {
    const d = new Date(t.date);
    if (d.getFullYear() === year && d.getMonth() === month) {
      const k = d.getDate();
      if (!byDay[k]) byDay[k] = [];
      byDay[k].push(t);
    }
  });

  const prev = () => { if (month === 0) { setMonth(11); setYear(y => y-1); } else setMonth(m => m-1); };
  const next = () => { if (month === 11) { setMonth(0); setYear(y => y+1); } else setMonth(m => m+1); };

  const cells = [];
  for (let i = 0; i < first; i++) cells.push(null);
  for (let d = 1; d <= days; d++) cells.push(d);

  return (
    <div>
      <div style={{ display:"flex", alignItems:"center", justifyContent:"space-between", marginBottom:16 }}>
        <button onClick={prev} style={btnSm}>‹</button>
        <span style={{ fontWeight:700, fontSize:15 }}>{MONTHS[month]} {year}</span>
        <button onClick={next} style={btnSm}>›</button>
      </div>
      <div style={{ display:"grid", gridTemplateColumns:"repeat(7,1fr)", gap:4 }}>
        {["Su","Mo","Tu","We","Th","Fr","Sa"].map(d =>
          <div key={d} style={{ textAlign:"center", fontSize:10, opacity:.4, padding:"4px 0", fontWeight:600 }}>{d}</div>
        )}
        {cells.map((d, i) => {
          if (!d) return <div key={`e${i}`} />;
          const ts = byDay[d] || [];
          const wins = ts.filter(t => parseFloat(t.pnl) > 0).length;
          const losses = ts.filter(t => parseFloat(t.pnl) < 0).length;
          const bg = ts.length === 0 ? "rgba(255,255,255,.03)"
            : wins > losses ? "rgba(16,185,129,.25)"
            : losses > wins ? "rgba(239,68,68,.2)"
            : "rgba(245,158,11,.2)";
          return (
            <div key={d} style={{ background:bg, borderRadius:6, padding:"6px 4px", textAlign:"center", fontSize:11, minHeight:36, cursor: ts.length ? "pointer" : "default", border: d === now.getDate() && month === now.getMonth() && year === now.getFullYear() ? `1px solid ${accent}` : "1px solid transparent" }}>
              <div style={{ fontWeight:600 }}>{d}</div>
              {ts.length > 0 && <div style={{ fontSize:9, opacity:.7 }}>{ts.length}t</div>}
            </div>
          );
        })}
      </div>
    </div>
  );
}

// ─── TRADE ROW ───────────────────────────────────────────────────────────────

function TradeRow({ t, accent, glow, S, onClick, onDelete }) {
  const pnl = parseFloat(t.pnl);
  const pnlColor = isNaN(pnl) ? "var(--text2)" : pnl > 0 ? "#22c55e" : pnl < 0 ? "#ef4444" : "var(--text2)";

  return (
    <div style={S.tradeRow} className="trade-row slide-in" onClick={onClick}>
      <div style={{ width:40, height:40, borderRadius:10, background:`rgba(${glow},.12)`, display:"flex", alignItems:"center", justifyContent:"center", fontSize:12, fontWeight:800, color:accent, flexShrink:0 }}>
        {t.asset?.split("/")[0]?.slice(0,3)}
      </div>
      <div style={{ flex:1, minWidth:0 }}>
        <div style={{ display:"flex", alignItems:"center", gap:8, marginBottom:4 }}>
          <span style={{ fontWeight:700, fontSize:14 }}>{t.asset}</span>
          <span style={S.pill(t.side === "Long" ? "#22c55e" : "#ef4444")}>{t.side}</span>
          {t.setup && <span style={S.pill(accent)}>{t.setup}</span>}
          {t.grade && <span style={{ ...S.pill(GRADE_COLOR[t.grade]||accent), fontFamily:"'JetBrains Mono', monospace" }}>{t.grade}</span>}
        </div>
        <div style={{ fontSize:11, opacity:.45 }}>{t.date?.replace("T"," ").slice(0,16)}</div>
      </div>
      <div style={{ textAlign:"right", flexShrink:0 }}>
        <div style={{ fontFamily:"'JetBrains Mono',monospace", fontWeight:700, fontSize:15, color:pnlColor }}>
          {isNaN(pnl) ? "—" : `${pnl >= 0 ? "+" : ""}$${pnl.toFixed(2)}`}
        </div>
        {t.rr && <div style={{ fontSize:11, opacity:.45 }}>R:R {t.rr}</div>}
      </div>
      <button onClick={e => { e.stopPropagation(); onDelete(); }}
        style={{ background:"rgba(239,68,68,.1)", border:"1px solid rgba(239,68,68,.2)", color:"#ef4444", borderRadius:7, padding:"6px 10px", cursor:"pointer", fontSize:12, flexShrink:0 }}>✕</button>
    </div>
  );
}

// ─── TOGGLE ──────────────────────────────────────────────────────────────────

function Toggle({ on, onChange, accent, glow }) {
  return (
    <div onClick={() => onChange(!on)} style={{ width:44, height:24, borderRadius:12, background: on ? accent : "rgba(255,255,255,.12)", cursor:"pointer", position:"relative", transition:"background .25s", flexShrink:0 }}>
      <div style={{ width:18, height:18, background:"white", borderRadius:"50%", position:"absolute", top:3, left: on ? 23 : 3, transition:"left .25s", boxShadow:"0 1px 4px rgba(0,0,0,.3)" }} />
    </div>
  );
}

// ─── TRADE FORM ──────────────────────────────────────────────────────────────

function TradeForm({ trade, assets, accent, glow, S, dark, onSave, onClose }) {
  const [t, setT] = useState(trade);
  const dragRef = useRef(null);

  const set = (k, v) => setT(prev => ({ ...prev, [k]: v }));
  const rr = calcRR(t.entry, t.sl, t.tp, t.side);
  const pnl = calcPnl(t.entry, t.exit, t.qty, t.side);

  const toggleEmotion = (e) => {
    setT(prev => ({ ...prev, emotions: prev.emotions?.includes(e) ? prev.emotions.filter(x => x !== e) : [...(prev.emotions||[]), e] }));
  };

  const handleImgDrop = (e) => {
    e.preventDefault();
    const files = Array.from(e.dataTransfer?.files || e.target.files || []);
    files.forEach(f => {
      const reader = new FileReader();
      reader.onload = ev => setT(prev => ({ ...prev, screenshots: [...(prev.screenshots||[]), ev.target.result] }));
      reader.readAsDataURL(f);
    });
  };

  const isEdit = !!trade.entry;

  return (
    <div style={S.modal} onClick={e => e.target === e.currentTarget && onClose()}>
      <div style={S.modalBox}>
        <div style={{ display:"flex", alignItems:"center", justifyContent:"space-between", marginBottom:24 }}>
          <div style={{ fontSize:18, fontWeight:800, letterSpacing:"-0.5px" }}>{isEdit ? "Edit Trade" : "Log New Trade"}</div>
          <button onClick={onClose} style={{ background:"rgba(255,255,255,.07)", border:"none", color:"var(--text)", width:32, height:32, borderRadius:8, cursor:"pointer", fontSize:16 }}>✕</button>
        </div>

        {/* BASIC DETAILS */}
        <div style={{ ...S.sectionHead }}>Basic Details</div>
        <div style={{ ...S.grid2, marginBottom:14 }} className="grid2">
          <div>
            <label style={S.label}>Asset / Pair</label>
            <select style={S.input} value={t.asset} onChange={e => set("asset", e.target.value)}>
              {assets.map(a => <option key={a} value={a}>{a}</option>)}
            </select>
          </div>
          <div>
            <label style={S.label}>Trade Side</label>
            <div style={{ display:"flex", gap:8 }}>
              {["Long","Short"].map(s => (
                <button key={s} onClick={() => set("side", s)} style={{ ...S.tag(t.side===s, s==="Long" ? "#22c55e" : "#ef4444"), flex:1, padding:"10px 8px", fontSize:13, borderRadius:8 }}>
                  {s === "Long" ? "▲ Long" : "▼ Short"}
                </button>
              ))}
            </div>
          </div>
        </div>

        <div style={{ ...S.grid3, marginBottom:14 }} className="grid3">
          {[["Entry Price","entry"],["Exit Price","exit"],["Qty / Lot Size","qty"]].map(([l,k]) => (
            <div key={k}>
              <label style={S.label}>{l}</label>
              <input style={{ ...S.input, fontFamily:"'JetBrains Mono',monospace" }} type="number" value={t[k]} onChange={e => set(k, e.target.value)} placeholder="0.00" />
            </div>
          ))}
        </div>

        <div style={{ ...S.grid3, marginBottom:14 }} className="grid3">
          {[["Stop Loss","sl"],["Take Profit","tp"]].map(([l,k]) => (
            <div key={k}>
              <label style={S.label}>{l}</label>
              <input style={{ ...S.input, fontFamily:"'JetBrains Mono',monospace" }} type="number" value={t[k]} onChange={e => set(k, e.target.value)} placeholder="0.00" />
            </div>
          ))}
          <div>
            <label style={S.label}>Date & Time</label>
            <input style={S.input} type="datetime-local" value={t.date} onChange={e => set("date", e.target.value)} />
          </div>
        </div>

        {/* RR Preview */}
        {(rr || pnl) && (
          <div style={{ display:"flex", gap:12, marginBottom:16 }}>
            {rr && <div style={{ background:`rgba(${glow},.1)`, border:`1px solid rgba(${glow},.2)`, borderRadius:8, padding:"10px 16px", flex:1, textAlign:"center" }}>
              <div style={{ fontSize:10, opacity:.5, textTransform:"uppercase", letterSpacing:".8px", marginBottom:4 }}>Risk:Reward</div>
              <div style={{ fontSize:20, fontWeight:800, color:accent, fontFamily:"'JetBrains Mono',monospace" }}>1 : {rr}</div>
            </div>}
            {pnl && <div style={{ background: parseFloat(pnl)>=0 ? "rgba(34,197,94,.1)" : "rgba(239,68,68,.1)", border: `1px solid ${parseFloat(pnl)>=0 ? "rgba(34,197,94,.2)" : "rgba(239,68,68,.2)"}`, borderRadius:8, padding:"10px 16px", flex:1, textAlign:"center" }}>
              <div style={{ fontSize:10, opacity:.5, textTransform:"uppercase", letterSpacing:".8px", marginBottom:4 }}>Est. P&L</div>
              <div style={{ fontSize:20, fontWeight:800, color: parseFloat(pnl)>=0 ? "#22c55e" : "#ef4444", fontFamily:"'JetBrains Mono',monospace" }}>{parseFloat(pnl)>=0 ? "+" : ""}${parseFloat(pnl).toFixed(2)}</div>
            </div>}
          </div>
        )}

        <div style={{ ...S.grid2, marginBottom:18 }} className="grid2">
          <div>
            <label style={S.label}>Setup Tag</label>
            <div style={{ display:"flex", flexWrap:"wrap", gap:6 }}>
              {SETUPS.map(s => <button key={s} onClick={() => set("setup", t.setup===s?"":s)} style={S.tag(t.setup===s)}>{s}</button>)}
            </div>
          </div>
          <div>
            <label style={S.label}>Trade Grade</label>
            <div style={{ display:"flex", gap:6 }}>
              {GRADES.map(g => <button key={g} onClick={() => set("grade", t.grade===g?"":g)} style={{ ...S.tag(t.grade===g, GRADE_COLOR[g]), fontFamily:"'JetBrains Mono',monospace", minWidth:34, padding:"6px 8px", fontSize:12 }}>{g}</button>)}
            </div>
          </div>
        </div>

        <div style={S.divider} />

        {/* PSYCHOLOGY */}
        <div style={S.sectionHead}>Trader Psychology</div>
        <label style={S.label}>Emotions During This Trade</label>
        <div style={{ display:"flex", flexWrap:"wrap", gap:7, marginBottom:16 }}>
          {EMOTIONS.map(e => (
            <button key={e} onClick={() => toggleEmotion(e)} style={{ ...S.tag(t.emotions?.includes(e), EMOTION_COLORS[e]), fontSize:12 }}>{e}</button>
          ))}
        </div>

        <div style={{ background:`rgba(${glow},.07)`, border:`1px solid rgba(${glow},.15)`, borderRadius:10, padding:"16px 18px", marginBottom:18 }}>
          <div style={{ fontSize:13, fontWeight:700, marginBottom:12, color:accent }}>
            ◉ Did you genuinely follow your price action system for this trade?
          </div>
          <div style={{ display:"flex", gap:10 }}>
            {["Yes","Partially","No"].map(opt => (
              <button key={opt} onClick={() => set("followedSystem", opt)}
                style={{ ...S.tag(t.followedSystem===opt, opt==="Yes"?"#22c55e":opt==="No"?"#ef4444":"#f59e0b"), flex:1, padding:"9px 8px", fontSize:12, borderRadius:8 }}>
                {opt==="Yes" ? "✓ Yes" : opt==="Partially" ? "~ Partially" : "✗ No"}
              </button>
            ))}
          </div>
        </div>

        <div style={S.divider} />

        {/* NOTES */}
        <div style={S.sectionHead}>Notes & Reflection</div>
        <textarea
          style={{ ...S.input, minHeight:110, resize:"vertical", lineHeight:1.6 }}
          placeholder="Trade reasoning, market structure, mistakes, emotions, lessons learned, improvements..."
          value={t.notes} onChange={e => set("notes", e.target.value)}
        />

        <div style={S.divider} />

        {/* SCREENSHOTS */}
        <div style={S.sectionHead}>Screenshots</div>
        <div
          onDrop={handleImgDrop} onDragOver={e => e.preventDefault()}
          style={{ border:`2px dashed rgba(${glow},.25)`, borderRadius:10, padding:"22px 16px", textAlign:"center", cursor:"pointer", marginBottom:14, background:`rgba(${glow},.03)` }}
          onClick={() => dragRef.current?.click()}
        >
          <div style={{ fontSize:24, marginBottom:6, opacity:.4 }}>⬆</div>
          <div style={{ fontSize:12, opacity:.45 }}>Drag & drop screenshots, or click to browse</div>
          <input ref={dragRef} type="file" accept="image/*" multiple style={{ display:"none" }} onChange={handleImgDrop} />
        </div>
        {t.screenshots?.length > 0 && (
          <div style={{ display:"flex", gap:10, flexWrap:"wrap", marginBottom:16 }}>
            {t.screenshots.map((src, i) => (
              <div key={i} style={{ position:"relative" }}>
                <img src={src} alt={`ss${i}`} style={{ width:110, height:72, objectFit:"cover", borderRadius:8, border:"1px solid var(--border)" }} />
                <button onClick={() => setT(prev => ({ ...prev, screenshots: prev.screenshots.filter((_,j) => j!==i) }))}
                  style={{ position:"absolute", top:3, right:3, background:"rgba(0,0,0,.7)", color:"white", border:"none", borderRadius:4, width:18, height:18, cursor:"pointer", fontSize:10 }}>✕</button>
              </div>
            ))}
          </div>
        )}

        <div style={{ display:"flex", gap:12, justifyContent:"flex-end", marginTop:8 }}>
          <button style={S.btnGhost} onClick={onClose}>Cancel</button>
          <button style={S.btn} className="btn-primary" onClick={() => onSave(t)}>Save Trade</button>
        </div>
      </div>
    </div>
  );
}

// ─── ANALYTICS PANEL ─────────────────────────────────────────────────────────

function AnalyticsPanel({ trades, accent, glow, S }) {
  const months = {};
  trades.forEach(t => {
    const d = new Date(t.date);
    const k = `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}`;
    if (!months[k]) months[k] = { pnl:0, count:0, wins:0 };
    months[k].pnl += parseFloat(t.pnl)||0;
    months[k].count++;
    if ((parseFloat(t.pnl)||0) > 0) months[k].wins++;
  });
  const monthKeys = Object.keys(months).sort().slice(-6);

  const gradeCount = {};
  trades.forEach(t => { if (t.grade) gradeCount[t.grade] = (gradeCount[t.grade]||0)+1; });

  const setupCount = {};
  trades.forEach(t => { if (t.setup) setupCount[t.setup] = (setupCount[t.setup]||0)+1; });

  return (
    <div style={{ display:"flex", flexDirection:"column", gap:20 }}>
      {/* Monthly PnL */}
      <div style={S.card}>
        <div style={S.sectionHead}>Monthly P&L</div>
        {monthKeys.length > 0 ? (
          <div style={{ display:"flex", gap:12, alignItems:"flex-end", height:100 }}>
            {monthKeys.map(k => {
              const m = months[k];
              const maxAbs = Math.max(...monthKeys.map(x => Math.abs(months[x].pnl)), 1);
              const h = Math.abs(m.pnl) / maxAbs * 80;
              return (
                <div key={k} style={{ flex:1, display:"flex", flexDirection:"column", alignItems:"center", gap:4 }}>
                  <div style={{ fontSize:10, opacity:.5, fontFamily:"'JetBrains Mono',monospace", color: m.pnl>=0?"#22c55e":"#ef4444" }}>
                    {m.pnl>=0?"+":""}{m.pnl.toFixed(0)}
                  </div>
                  <div style={{ width:"100%", height:h||4, background: m.pnl>=0?"rgba(34,197,94,.35)":"rgba(239,68,68,.35)", borderRadius:4, border:`1px solid ${m.pnl>=0?"rgba(34,197,94,.5)":"rgba(239,68,68,.5)"}`, minHeight:4 }} />
                  <div style={{ fontSize:10, opacity:.4 }}>{k.slice(5)}</div>
                </div>
              );
            })}
          </div>
        ) : <div style={{ opacity:.35, fontSize:13 }}>No data yet</div>}
      </div>

      <div style={{ ...S.grid2 }} className="grid2">
        {/* Grade breakdown */}
        <div style={S.card}>
          <div style={S.sectionHead}>Trade Grades</div>
          <div style={{ display:"flex", flexDirection:"column", gap:9 }}>
            {GRADES.map(g => {
              const c = gradeCount[g] || 0;
              const total = trades.length || 1;
              const pct = Math.round(c/total*100);
              return (
                <div key={g} style={{ display:"flex", alignItems:"center", gap:10 }}>
                  <span style={{ fontFamily:"'JetBrains Mono',monospace", fontWeight:700, color:GRADE_COLOR[g], width:22, textAlign:"center", fontSize:12 }}>{g}</span>
                  <div style={{ flex:1, height:8, background:"rgba(255,255,255,.06)", borderRadius:4, overflow:"hidden" }}>
                    <div style={{ width:`${pct}%`, height:"100%", background:GRADE_COLOR[g], borderRadius:4, transition:"width .4s" }} />
                  </div>
                  <span style={{ fontSize:11, opacity:.4, width:28, textAlign:"right" }}>{c}</span>
                </div>
              );
            })}
          </div>
        </div>

        {/* Setup breakdown */}
        <div style={S.card}>
          <div style={S.sectionHead}>Setup Types</div>
          <div style={{ display:"flex", flexDirection:"column", gap:9 }}>
            {SETUPS.map(s => {
              const c = setupCount[s] || 0;
              const total = trades.length || 1;
              const pct = Math.round(c/total*100);
              return (
                <div key={s} style={{ display:"flex", alignItems:"center", gap:10 }}>
                  <span style={{ fontSize:11, width:100, opacity:.7, flexShrink:0 }}>{s}</span>
                  <div style={{ flex:1, height:8, background:"rgba(255,255,255,.06)", borderRadius:4, overflow:"hidden" }}>
                    <div style={{ width:`${pct}%`, height:"100%", background:accent, borderRadius:4, transition:"width .4s", opacity:.8 }} />
                  </div>
                  <span style={{ fontSize:11, opacity:.4, width:28, textAlign:"right" }}>{c}</span>
                </div>
              );
            })}
          </div>
        </div>
      </div>
    </div>
  );
}

// ─── PSYCHOLOGY PANEL ────────────────────────────────────────────────────────

function PsychologyPanel({ trades, accent, glow, S }) {
  const emotionCount = {};
  EMOTIONS.forEach(e => { emotionCount[e] = 0; });
  trades.forEach(t => t.emotions?.forEach(e => { emotionCount[e] = (emotionCount[e]||0)+1; }));

  const totalEmotionLogs = Object.values(emotionCount).reduce((a,b) => a+b, 0) || 1;

  const followCount = { Yes:0, Partially:0, No:0 };
  trades.forEach(t => { if (t.followedSystem) followCount[t.followedSystem]++; });

  const disciplineScore = trades.length
    ? Math.round(followCount.Yes / trades.length * 100) : 0;

  return (
    <div style={{ display:"flex", flexDirection:"column", gap:20 }}>
      {/* Discipline score */}
      <div style={{ ...S.card, background:`linear-gradient(135deg,rgba(${glow},.1),rgba(${glow},.03))`, border:`1px solid rgba(${glow},.2)` }}>
        <div style={{ display:"flex", alignItems:"center", gap:24 }}>
          <Donut pct={disciplineScore} color={accent} size={96} />
          <div>
            <div style={{ fontSize:13, fontWeight:600, opacity:.5, textTransform:"uppercase", letterSpacing:"1px", marginBottom:8 }}>Discipline Score</div>
            <div style={{ fontSize:36, fontWeight:900, color:accent, fontFamily:"'Syne',sans-serif" }}>{disciplineScore}%</div>
            <div style={{ fontSize:12, opacity:.4, marginTop:4 }}>Followed system: {followCount.Yes}× / {trades.length} trades</div>
          </div>
          <div style={{ marginLeft:"auto", display:"flex", gap:12 }}>
            {[["Yes","#22c55e"],["Partially","#f59e0b"],["No","#ef4444"]].map(([k,c]) => (
              <div key={k} style={{ textAlign:"center" }}>
                <div style={{ fontSize:22, fontWeight:800, color:c, fontFamily:"'Syne',sans-serif" }}>{followCount[k]}</div>
                <div style={{ fontSize:11, opacity:.4 }}>{k}</div>
              </div>
            ))}
          </div>
        </div>
      </div>

      {/* Emotion heatmap */}
      <div style={S.card}>
        <div style={S.sectionHead}>Emotional Frequency</div>
        <div style={{ display:"flex", flexDirection:"column", gap:10 }}>
          {EMOTIONS.sort((a,b) => emotionCount[b]-emotionCount[a]).map(e => {
            const pct = Math.round(emotionCount[e]/totalEmotionLogs*100);
            return (
              <div key={e} style={{ display:"flex", alignItems:"center", gap:12 }}>
                <div style={{ width:14, height:14, borderRadius:4, background:EMOTION_COLORS[e], flexShrink:0 }} />
                <span style={{ fontSize:12, width:130, flexShrink:0 }}>{e}</span>
                <div style={{ flex:1, height:10, background:"rgba(255,255,255,.06)", borderRadius:5, overflow:"hidden" }}>
                  <div style={{ width:`${pct}%`, height:"100%", background:EMOTION_COLORS[e], borderRadius:5, opacity:.75, transition:"width .5s" }} />
                </div>
                <span style={{ fontSize:12, fontFamily:"'JetBrains Mono',monospace", opacity:.5, width:32, textAlign:"right" }}>{emotionCount[e]}</span>
              </div>
            );
          })}
        </div>
      </div>
    </div>
  );
}
